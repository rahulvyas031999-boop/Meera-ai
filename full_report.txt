fun rmsOf(s: ShortArray): Float {
    if (s.isEmpty()) return 0f
    var sum = 0.0
    for (v in s) { val f = v / 32768.0; sum += f * f }
    return sqrt(sum / s.size).toFloat()
}

fun ShortArray.toLeBytes(): ByteArray {
    val out = ByteArray(size * 2)
    for (i in indices) {
        out[i * 2] = (this[i].toInt() and 0xFF).toByte()
        out[i * 2 + 1] = ((this[i].toInt() shr 8) and 0xFF).toByte()
    }
    return out
}

fun joinPcm(list: List<ShortArray>): ByteArray {
    val total = list.sumOf { it.size }
    val out = ByteArray(total * 2)
    var o = 0
    for (f in list) for (s in f) {
        out[o++] = (s.toInt() and 0xFF).toByte()
        out[o++] = ((s.toInt() shr 8) and 0xFF).toByte()
    }
    return out
}

fun pcmToWav(pcm: ByteArray, rate: Int): ByteArray {
    val h = ByteArray(44)
    fun i32(o: Int, v: Int) { h[o] = v.toByte(); h[o + 1] = (v shr 8).toByte(); h[o + 2] = (v shr 16).toByte(); h[o + 3] = (v shr 24).toByte() }
    fun i16(o: Int, v: Int) { h[o] = v.toByte(); h[o + 1] = (v shr 8).toByte() }
    "RIFF".toByteArray().copyInto(h, 0)
    i32(4, 36 + pcm.size)
    "WAVE".toByteArray().copyInto(h, 8)
    "fmt ".toByteArray().copyInto(h, 12)
    i32(16, 16); i16(20, 1); i16(22, 1); i32(24, rate); i32(28, rate * 2); i16(32, 2); i16(34, 16)
    "data".toByteArray().copyInto(h, 36)
    i32(40, pcm.size)
    return h + pcm
}

class MicRecorder(private val onFrame: (ShortArray) -> Unit) {
    private var rec: AudioRecord? = null
    private var thread: Thread? = null
    @Volatile private var running = false
    private val fx = mutableListOf<AudioEffect>()

    @SuppressLint("MissingPermission")
    fun start() {
        val min = AudioRecord.getMinBufferSize(Cfg.IN_RATE, AudioFormat.CHANNEL_IN_MONO, AudioFormat.ENCODING_PCM_16BIT)
        val r = AudioRecord(
            MediaRecorder.AudioSource.VOICE_COMMUNICATION, Cfg.IN_RATE,
            AudioFormat.CHANNEL_IN_MONO, AudioFormat.ENCODING_PCM_16BIT,
            maxOf(min, Cfg.FRAME_SAMPLES * 2 * 6)
        )
        if (r.state != AudioRecord.STATE_INITIALIZED) {
            r.release()
            throw IllegalStateException("माइक शुरू नहीं हो पाया")
        }
        val sid = r.audioSessionId
        try {
            if (AcousticEchoCanceler.isAvailable()) AcousticEchoCanceler.create(sid)?.let { it.enabled = true; fx += it }
            if (NoiseSuppressor.isAvailable()) NoiseSuppressor.create(sid)?.let { it.enabled = true; fx += it }
            if (AutomaticGainControl.isAvailable()) AutomaticGainControl.create(sid)?.let { it.enabled = true; fx += it }
        } catch (_: Throwable) {}
        rec = r
        running = true
        r.startRecording()
        thread = Thread({
            android.os.Process.setThreadPriority(android.os.Process.THREAD_PRIORITY_URGENT_AUDIO)
            val buf = ShortArray(Cfg.FRAME_SAMPLES)
            while (running) {
                val n = r.read(buf, 0, buf.size)
                if (n > 0) onFrame(buf.copyOf(n)) else if (n < 0) break
            }
        }, "meera-mic").also { it.isDaemon = true; it.start() }
    }

    fun stop() {
        running = false
        try { rec?.stop() } catch (_: Exception) {}
        if (Thread.currentThread() != thread) try { thread?.join(300) } catch (_: Exception) {}
        fx.forEach { try { it.release() } catch (_: Exception) {} }
        fx.clear()
        try { rec?.release() } catch (_: Exception) {}
        rec = null
    }
}

class PcmPlayer {
    private class Chunk(val data: ByteArray, val gen: Int)

    private val track: AudioTrack
    private val queue = LinkedBlockingQueue<Chunk>()
    private val pending = AtomicInteger(0)
    @Volatile private var gen = 0
    @Volatile private var playUntil = 0L
    @Volatile private var alive = true
    @Volatile var level = 0f
        private set
    private val thread: Thread

    val busy: Boolean get() = pending.get() > 0 || System.nanoTime() < playUntil

    init {
        val rate = Cfg.OUT_RATE
        val min = AudioTrack.getMinBufferSize(rate, AudioFormat.CHANNEL_OUT_MONO, AudioFormat.ENCODING_PCM_16BIT)
        track = AudioTrack.Builder()
            .setAudioAttributes(
                AudioAttributes.Builder()
                    .setUsage(AudioAttributes.USAGE_VOICE_COMMUNICATION)
                    .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH)
                    .build()
            )
            .setAudioFormat(
                AudioFormat.Builder()
                    .setSampleRate(rate)
                    .setChannelMask(AudioFormat.CHANNEL_OUT_MONO)
                    .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
                    .build()
            )
            .setBufferSizeInBytes(maxOf(min, rate * 2))
            .setTransferMode(AudioTrack.MODE_STREAM)
            .build()
        track.play()
        thread = Thread { loop() }.also { it.isDaemon = true; it.start() }
    }

    private fun loop() {
        val slice = Cfg.OUT_RATE * 2 / 50 // 20ms
        try {
            while (alive) {
                val c = queue.poll(100, TimeUnit.MILLISECONDS) ?: continue
                var off = 0
                while (off < c.data.size && alive && c.gen == gen) {
                    val n = minOf(slice, c.data.size - off)
                    level = levelOf(c.data, off, n)
                    val w = track.write(c.data, off, n)
                    if (w < 0) break
                    off += w
                    playUntil = maxOf(playUntil, System.nanoTime()) + w * 1_000_000_000L / (Cfg.OUT_RATE * 2)
                }
                if (c.gen == gen) pending.addAndGet(-c.data.size)
            }
        } catch (_: InterruptedException) {
        }
    }

    private fun levelOf(b: ByteArray, off: Int, n: Int): Float {
        var s = 0.0
        var cnt = 0
        var i = off
        while (i + 1 < off + n) {
            val v = ((b[i + 1].toInt() shl 8) or (b[i].toInt() and 0xFF)).toShort().toInt() / 32768.0
            s += v * v; cnt++; i += 2
        }
        return if (cnt == 0) 0f else (sqrt(s / cnt) * 5).toFloat().coerceIn(0f, 1f)
    }

    @Synchronized
    fun enqueue(pcm: ByteArray) {
        if (pcm.isEmpty()) return
        pending.addAndGet(pcm.size)
        queue.put(Chunk(pcm, gen))
    }

    @Synchronized
    fun flush() {
        gen++
        queue.clear()
        pending.set(0)
        playUntil = 0L
        level = 0f
        try { track.pause(); track.flush(); track.play() } catch (_: Exception) {}
    }

    fun release() {
        alive = false
        thread.interrupt()
        try { track.stop() } catch (_: Exception) {}
        try { track.release() } catch (_: Exception) {}
    }
}

class TurnDetector(private val silenceMs: () -> Int) {
    enum class Ev { NONE, SPEECH_START, SPEECH_END, NOISE }

    private var floor = 0.004f
    private var speaking = false
    private var onset = 0
    private var voicedMs = 0
    private var silentMs = 0

    @Synchronized fun reset() { speaking = false; onset = 0; voicedMs = 0; silentMs = 0 }

    @Synchronized fun forceSpeaking() { speaking = true; onset = 0; voicedMs = Cfg.MIN_SPEECH_MS; silentMs = 0 }

    @Synchronized
    fun process(rms: Float): Ev {
        val threshold = maxOf(Cfg.VAD_MIN_RMS, floor * Cfg.VAD_NOISE_RATIO)
        val voiced = rms > threshold
        if (!speaking) {
            if (voiced) {
                onset++
                if (onset >= Cfg.SPEECH_START_FRAMES) {
                    speaking = true
                    voicedMs = onset * Cfg.FRAME_MS
                    silentMs = 0
                    return Ev.SPEECH_START
                }
            } else {
                onset = 0
                floor = (floor * 0.95f + rms * 0.05f).coerceIn(0.001f, 0.03f)
            }
            return Ev.NONE
        }
        if (rms > threshold * 0.75f) {
            voicedMs += Cfg.FRAME_MS
            silentMs = 0
        } else {
            silentMs += Cfg.FRAME_MS
        }
        if (silentMs >= silenceMs()) {
            val ok = voicedMs >= Cfg.MIN_SPEECH_MS
            speaking = false; onset = 0; voicedMs = 0; silentMs = 0
            return if (ok) Ev.SPEECH_END else Ev.NOISE
        }
        return Ev.NONE
    }
}

package com.meraai.app

import okhttp3.OkHttpClient
import okhttp3.Request
import okhttp3.Response

import okhttp3.WebSocketListener
import okio.ByteString
import org.json.JSONArray

class GeminiLive(private val http: OkHttpClient, private val ev: Events) {
    interface Events {
        fun onReady()
        fun onAudio(pcm: ByteArray)
        fun onInputText(t: String)
        fun onOutputText(t: String)
        fun onToolCall(id: String, name: String, args: JSONObject)
        fun onInterrupted()
        fun onTurnComplete()
        fun onError(msg: String)
    }

    private var ws: WebSocket? = null
    @Volatile var ready = false
        private set
    @Volatile private var closedByUs = false

    fun connect(key: String, model: String, voice: String, system: String, silenceMs: Int) {
        closedByUs = false
        ready = false
        val url = "wss://generativelanguage.googleapis.com/ws/google.ai.generativelanguage.v1beta.GenerativeService.BidiGenerateContent?key=$key"
        val setup = JSONObject().put(
            "setup", JSONObject()
                .put("model", "models/$model")
                .put(
                    "generationConfig", JSONObject()
                        .put("responseModalities", JSONArray().put("AUDIO"))
                        .put(
                            "speechConfig", JSONObject().put(
                                "voiceConfig", JSONObject().put(
                                    "prebuiltVoiceConfig", JSONObject().put("voiceName", voice)
                                )
                            )
                        )
                )
                .put("systemInstruction", JSONObject().put("parts", JSONArray().put(JSONObject().put("text", system))))
                .put("tools", MeeraTools.getGeminiToolsJson())
                .put(
                    "realtimeInputConfig", JSONObject().put(
                        "automaticActivityDetection", JSONObject()
                            .put("disabled", false)
                            .put("startOfSpeechSensitivity", "START_SENSITIVITY_LOW")
                            .put("endOfSpeechSensitivity", "END_SENSITIVITY_LOW")
                            .put("prefixPaddingMs", 100)
                            .put("silenceDurationMs", silenceMs)
                    )
                )
                .put("inputAudioTranscription", JSONObject())
                .put("outputAudioTranscription", JSONObject())
        )
        ws = http.newWebSocket(Request.Builder().url(url).build(), object : WebSocketListener() {
            override fun onOpen(webSocket: WebSocket, response: Response) {
                webSocket.send(setup.toString())
            }

            override fun onMessage(webSocket: WebSocket, text: String) = handle(text)
            override fun onMessage(webSocket: WebSocket, bytes: ByteString) = handle(bytes.utf8())

            override fun onClosing(webSocket: WebSocket, code: Int, reason: String) {
                webSocket.close(1000, null)
                if (!closedByUs) ev.onError("Live सेशन बंद हुआ ($code) $reason".trim())
            }

            override fun onFailure(webSocket: WebSocket, t: Throwable, response: Response?) {
                if (closedByUs) return
                val c = response?.code
                ev.onError(
                    when (c) {
                        401, 403 -> "API Key सही नहीं है या Live की इजाज़त नहीं (HTTP $c)"
                        null -> "कनेक्शन फेल: ${t.message ?: t.javaClass.simpleName}"
                        else -> "Live कनेक्शन फेल (HTTP $c)"
                    }
                )
            }
        })
    }

    private fun handle(s: String) {
        try {
            val j = JSONObject(s)
            if (j.has("setupComplete")) { ready = true; ev.onReady(); return }
            j.optJSONObject("error")?.let { ev.onError(it.optString("message", "Live API error")); return }

            // Handle Live Tool Call (e.g., open_app, read_screen, whatsapp_message)
            j.optJSONObject("toolCall")?.optJSONArray("functionCalls")?.let { calls ->
                for (i in 0 until calls.length()) {
                    val call = calls.optJSONObject(i) ?: continue
                    val id = call.optString("id", "")
                    val name = call.optString("name", "")
                    val args = call.optJSONObject("args") ?: JSONObject()
                    if (name.isNotBlank()) ev.onToolCall(id, name, args)
                }
            }

            val sc = j.optJSONObject("serverContent") ?: return
            if (sc.optBoolean("interrupted")) ev.onInterrupted()
            sc.optJSONObject("inputTranscription")?.optString("text")?.takeIf { it.isNotEmpty() }?.let { ev.onInputText(it) }
            sc.optJSONObject("outputTranscription")?.optString("text")?.takeIf { it.isNotEmpty() }?.let { ev.onOutputText(it) }
            val parts = sc.optJSONObject("modelTurn")?.optJSONArray("parts")
            if (parts != null) {
                for (i in 0 until parts.length()) {
                    val d = parts.optJSONObject(i)?.optJSONObject("inlineData")?.optString("data")
                    if (!d.isNullOrEmpty()) ev.onAudio(unb64(d))
                }
            }
            if (sc.optBoolean("turnComplete")) ev.onTurnComplete()
        } catch (_: Exception) {
        }
    }

    fun sendToolResponse(id: String, name: String, resultText: String) {
        if (!ready) return
        val respObj = JSONObject().put(
            "toolResponse", JSONObject().put(
                "functionResponses", JSONArray().put(
                    JSONObject()
                        .put("id", id)
                        .put("name", name)
                        .put("response", JSONObject().put("result", resultText))
                )
            )
        )
        ws?.send(respObj.toString())
    }

    fun sendAudio(pcm: ByteArray) {
        if (!ready) return
        val m = JSONObject().put(
            "realtimeInput", JSONObject().put(
                "audio", JSONObject().put("data", b64(pcm)).put("mimeType", "audio/pcm;rate=16000")
            )
        )
        ws?.send(m.toString())
    }

    fun close() {
        closedByUs = true
        ready = false
        try { ws?.close(1000, null) } catch (_: Exception) {}
        ws = null
    }
}

package com.meraai.app

import kotlinx.coroutines.CancellationException
import kotlinx.coroutines.Dispatchers

import okhttp3.Call
import okhttp3.Callback
import okhttp3.MediaType.Companion.toMediaType
import okhttp3.OkHttpClient
import okhttp3.Request

import org.json.JSONArray
import org.json.JSONObject

class GeminiRest(private val http: OkHttpClient, private val key: () -> String) {
    private val base = "https://generativelanguage.googleapis.com/v1beta/models"
    private val jsonType = "application/json".toMediaType()

    private fun request(model: String, method: String, body: JSONObject, sse: Boolean = false): Request {
        val q = if (sse) "?alt=sse" else ""
        return Request.Builder()
            .url("$base/$model:$method$q")
            .header("x-goog-api-key", key())
            .post(body.toString().toRequestBody(jsonType))
            .build()
    }

    private suspend fun Call.await(): Response = suspendCancellableCoroutine { c ->
        c.invokeOnCancellation { cancel() }
        enqueue(object : Callback {
            override fun onFailure(call: Call, e: IOException) { if (c.isActive) c.resumeWithException(e) }
            override fun onResponse(call: Call, response: Response) { c.resume(response) }
        })
    }

    private fun Response.okOrThrow(): Response {
        if (!isSuccessful) {
            val t = try { body?.string()?.take(300) ?: "" } catch (_: Exception) { "" }
            val c = code
            close()
            throw IOException("HTTP $c $t")
        }
        return this
    }

    private fun textOf(j: JSONObject): String {
        val parts = j.optJSONArray("candidates")?.optJSONObject(0)?.optJSONObject("content")?.optJSONArray("parts") ?: return ""
        val sb = StringBuilder()
        for (i in 0 until parts.length()) sb.append(parts.optJSONObject(i)?.optString("text", "") ?: "")
        return sb.toString()
    }

    // ---------- 1. STT (With Cough / Distant TV / Noise Reject Prompt) ----------
    private val sttPrompt = "Transcribe ONLY clear near-field human speech spoken directly to the assistant (Hindi in Devanagari, Hinglish, or English). " +
        "IMPORTANT: If the audio is a cough, throat clearing, sneeze, laugh, background TV/mobile sound, or indistinct noise, set hasSpeech to false and transcript to an empty string."

    suspend fun transcribe(wav: ByteArray, preferredStt: String = Cfg.STT_MODELS.first()): String = withContext(Dispatchers.IO) {
        var last: Exception? = null
        val sttList = (listOf(preferredStt) + Cfg.STT_MODELS).distinct()
        for ((idx, m) in sttList.withIndex()) {
            try {
                val structured = m != "gemini-3.5-transcribe"
                val promptText = if (structured) sttPrompt else "Transcribe clear spoken words only. Ignore coughs or background noise."
                val body = JSONObject().put(
                    "contents", JSONArray().put(
                        JSONObject().put("role", "user").put(
                            "parts", JSONArray()
                                .put(JSONObject().put("inlineData", JSONObject().put("mimeType", "audio/wav").put("data", b64(wav))))
                                .put(JSONObject().put("text", promptText))
                        )
                    )
                )
                if (structured) {
                    body.put(
                        "generationConfig", JSONObject()
                            .put("responseMimeType", "application/json")
                            .put(
                                "responseSchema", JSONObject()
                                    .put("type", "OBJECT")
                                    .put(
                                        "properties", JSONObject()
                                            .put("hasSpeech", JSONObject().put("type", "BOOLEAN"))
                                            .put("transcript", JSONObject().put("type", "STRING"))
                                    )
                                    .put("required", JSONArray().put("hasSpeech").put("transcript"))
                            )
                    )
                }
                val resp = http.newCall(request(m, "generateContent", body)).await().okOrThrow()
                val raw = resp.use { textOf(JSONObject(it.body!!.string())) }
                val text = if (structured) parseStt(raw) else raw.trim()
                if (text.isNotBlank()) return@withContext text
                if (idx == 0) return@withContext ""
            } catch (e: CancellationException) {
                throw e
            } catch (e: Exception) {
                last = e
            }
        }
        last?.let { throw it }
        ""
    }

    private fun parseStt(raw: String): String = try {
        val j = JSONObject(raw)
        if (j.optBoolean("hasSpeech", true)) j.optString("transcript", "").trim() else ""
    } catch (_: Exception) {
        raw.trim().trim('"')
    }

    // ---------- 2. LLM (Streaming with Preferred Brain & Thinking Level) ----------
    private fun sentenceEnd(b: CharSequence): Int {
        for (i in b.indices) {
            val c = b[i]
            if (c == '\n' || c == '।') { if (i >= 12) return i + 1 }
            else if ((c == '.' || c == '!' || c == '?' || c == ';') && i >= 65 && i + 1 < b.length && b[i + 1].isWhitespace()) return i + 1
        }
        return -1
    }

    suspend fun streamChat(
        history: List<Pair<String, String>>,
        system: String,
        preferredModel: String = Cfg.CHAT_MODELS.first(),
        onSentence: suspend (String) -> Unit,
    ): Unit = withContext<Unit>(Dispatchers.IO) {
        var last: Exception? = null
        val isDeep = preferredModel == "gemini-3.8-flash-thinking"
        val firstModel = if (isDeep) "gemini-3.8-flash" else preferredModel
        val modelList = (listOf(firstModel) + Cfg.CHAT_MODELS).distinct()

        for (m in modelList) {
            val levels = when {
                isDeep && m == firstModel -> listOf("HIGH", "LOW", null)
                m.contains("3.8") || m.contains("3.7") || m.contains("3.6") -> listOf("LOW", null)
                m.contains("3.1") -> listOf("MINIMAL", null)
                else -> listOf(null)
            }
            for (lv in levels) {
                var emitted = false
                val contents = JSONArray()
                for ((role, text) in history) {
                    contents.put(JSONObject().put("role", role).put("parts", JSONArray().put(JSONObject().put("text", text))))
                }
                val body = JSONObject()
                    .put("systemInstruction", JSONObject().put("parts", JSONArray().put(JSONObject().put("text", system))))
                    .put("contents", contents)
                if (lv != null) body.put("generationConfig", JSONObject().put("thinkingConfig", JSONObject().put("thinkingLevel", lv)))
                val call = http.newCall(request(m, "streamGenerateContent", body, sse = true))
                val handle = currentCoroutineContext().job.invokeOnCompletion { call.cancel() }
                try {
                    val resp = call.await().okOrThrow()
                    resp.use { r ->
                        val src = r.body!!.source()
                        val buf = StringBuilder()
                        while (true) {
                            currentCoroutineContext().ensureActive()
                            val line = src.readUtf8Line() ?: break
                            if (!line.startsWith("data:")) continue
                            val payload = line.substring(5).trim()
                            if (payload.isEmpty() || payload == "[DONE]") continue
                            val piece = textOf(JSONObject(payload))
                            if (piece.isEmpty()) continue
                            buf.append(piece)
                            while (true) {
                                val cut = sentenceEnd(buf)
                                if (cut < 0) break
                                val s = buf.substring(0, cut).trim()
                                buf.delete(0, cut)
                                if (s.isNotEmpty()) { emitted = true; onSentence(s) }
                            }
                        }
                        val rest = buf.toString().trim()
                        if (rest.isNotEmpty()) { emitted = true; onSentence(rest) }
                    }
                    return@withContext
                } catch (e: CancellationException) {
                    throw e
                } catch (e: Exception) {
                    if (emitted) throw e
                    last = e
                } finally {
                    handle.dispose()
                }
            }
        }
        throw (last ?: IOException("Chat मॉडल जवाब नहीं दे पाया"))
    }

    // ---------- 3. Expressive TTS (Persona Style Directing + Glitch-Free WAV Parser) ----------
    suspend fun tts(
        text: String,
        voice: String,
        preferredTts: String = Cfg.TTS_MODELS.first(),
        persona: String = "Meera",
        gfMode: Boolean = true
    ): ByteArray = withContext(Dispatchers.IO) {
        var last: Exception? = null
        val stylePrompt = when (persona) {
            "Jarvis" -> "crisp, composed, intelligent, confident, and sharp"
            "Rudra" -> "deep, bold, commanding, fearless, and grounded"
            else -> if (gfMode) "warm, playful, affectionate, expressive, and natural like a close companion" else "warm, friendly, natural, and conversational"
        }
        val ttsList = (listOf(preferredTts) + Cfg.TTS_MODELS).distinct()
        for (ttsModel in ttsList) {
            val styles = if (ttsModel.startsWith("gemini-3.8")) listOf(true, false) else listOf(false)
            for (withStyle in styles) {
                try {
                    val part = JSONObject().put("text", text)
                    if (withStyle) part.put("speechMetadata", JSONObject().put("style", stylePrompt))
                    val body = JSONObject()
                        .put("contents", JSONArray().put(JSONObject().put("role", "user").put("parts", JSONArray().put(part))))
                        .put(
                            "generationConfig", JSONObject()
                                .put("responseModalities", JSONArray().put("AUDIO"))
                                .put(
                                    "speechConfig", JSONObject().put(
                                        "voiceConfig", JSONObject().put(
                                            "prebuiltVoiceConfig", JSONObject().put("voiceName", voice)
                                        )
                                    )
                                )
                        )
                    val resp = http.newCall(request(ttsModel, "generateContent", body)).await().okOrThrow()
                    val j = resp.use { JSONObject(it.body!!.string()) }
                    val inline = j.optJSONArray("candidates")?.optJSONObject(0)?.optJSONObject("content")
                        ?.optJSONArray("parts")?.optJSONObject(0)?.optJSONObject("inlineData")
                        ?: throw IOException("TTS से ऑडियो नहीं मिला")
                    val rawBytes = unb64(inline.getString("data"))
                    val parsed = extractCleanPcm(rawBytes, inline.optString("mimeType", ""))
                    val pcm = if (parsed.second == Cfg.OUT_RATE) parsed.first else resample(parsed.first, parsed.second, Cfg.OUT_RATE)
                    return@withContext applyMicroFade(pcm)
                } catch (e: CancellationException) {
                    throw e
                } catch (e: Exception) {
                    last = e
                }
            }
        }
        throw (last ?: IOException("TTS फेल"))
    }

    private fun extractCleanPcm(bytes: ByteArray, mime: String): Pair<ByteArray, Int> {
        var rate = Regex("rate=(\\d+)").find(mime)?.groupValues?.get(1)?.toIntOrNull() ?: Cfg.OUT_RATE
        if (bytes.size > 44 && bytes[0] == 'R'.code.toByte() && bytes[1] == 'I'.code.toByte() && bytes[2] == 'F'.code.toByte() && bytes[3] == 'F'.code.toByte()) {
            rate = (bytes[24].toInt() and 0xFF) or ((bytes[25].toInt() and 0xFF) shl 8) or
                ((bytes[26].toInt() and 0xFF) shl 16) or ((bytes[27].toInt() and 0xFF) shl 24)
            var pos = 12
            while (pos + 8 <= bytes.size) {
                val id0 = bytes[pos].toInt().toChar()
                val id1 = bytes[pos + 1].toInt().toChar()
                val id2 = bytes[pos + 2].toInt().toChar()
                val id3 = bytes[pos + 3].toInt().toChar()
                val chunkSize = (bytes[pos + 4].toInt() and 0xFF) or ((bytes[pos + 5].toInt() and 0xFF) shl 8) or
                    ((bytes[pos + 6].toInt() and 0xFF) shl 16) or ((bytes[pos + 7].toInt() and 0xFF) shl 24)
                val dataStart = pos + 8
                if (id0 == 'd' && id1 == 'a' && id2 == 't' && id3 == 'a') {
                    val end = minOf(bytes.size, if (chunkSize > 0) dataStart + chunkSize else bytes.size)
                    val evenEnd = dataStart + ((end - dataStart) and -2)
                    return bytes.copyOfRange(dataStart, evenEnd) to rate
                }
                if (chunkSize < 0) break
                pos = dataStart + chunkSize + (chunkSize and 1)
            }
            val evenEnd = 44 + ((bytes.size - 44) and -2)
            return bytes.copyOfRange(44, evenEnd) to rate
        }
        val evenSize = bytes.size and -2
        return (if (evenSize == bytes.size) bytes else bytes.copyOfRange(0, evenSize)) to rate
    }

    private fun applyMicroFade(pcm: ByteArray): ByteArray {
        val samples = pcm.size / 2
        val fade = minOf(120, samples / 10)
        if (fade <= 2) return pcm
        for (i in 0 until fade) {
            val gain = i.toFloat() / fade
            val idx = i * 2
            val s = (((pcm[idx + 1].toInt() shl 8) or (pcm[idx].toInt() and 0xFF)).toShort() * gain).toInt()
            pcm[idx] = (s and 0xFF).toByte()
            pcm[idx + 1] = ((s shr 8) and 0xFF).toByte()

            val rIdx = (samples - 1 - i) * 2
            val rs = (((pcm[rIdx + 1].toInt() shl 8) or (pcm[rIdx].toInt() and 0xFF)).toShort() * gain).toInt()
            pcm[rIdx] = (rs and 0xFF).toByte()
            pcm[rIdx + 1] = ((rs shr 8) and 0xFF).toByte()
        }
        return pcm
    }

    private fun resample(pcm: ByteArray, from: Int, to: Int): ByteArray {
        val n = pcm.size / 2
        if (n < 2) return pcm
        val outN = (n.toLong() * to / from).toInt()
        val out = ByteArray(outN * 2)
        fun s(i: Int): Int {
            val k = i.coerceIn(0, n - 1)
            return ((pcm[k * 2 + 1].toInt() shl 8) or (pcm[k * 2].toInt() and 0xFF)).toShort().toInt()
        }
        for (o in 0 until outN) {
            val pos = o.toDouble() * from / to
            val i = pos.toInt()
            val f = pos - i
            val v = (s(i) * (1 - f) + s(i + 1) * f).toInt()
            out[o * 2] = v.toByte()
            out[o * 2 + 1] = (v shr 8).toByte()
        }
        return out
    }

    // ---------- 4. Background Memory Fact Extractor (Maya remember_fact style) ----------
    suspend fun extractMemoryFact(userText: String, existing: List<String>): String? = withContext(Dispatchers.IO) {
        if (userText.length < 6) return@withContext null
        try {
            val prompt = """
Analyze what the user just said: "$userText"
Already saved facts: ${existing.takeLast(15)}
Task: Did the user reveal a permanent personal fact worth remembering for months (their name, family/friends, job/project, city, favorite things, or a promise/plan)?
Do NOT save temporary mood, weather, greetings, or general questions.
If YES and not already saved, reply with ONLY 1 short clear sentence in Hinglish (e.g. "Boss ka naam Rahul hai", "Unka bhai Pune me rehta hai").
If NO, reply with ONLY: NONE
""".trimIndent()
            val body = JSONObject().put(
                "contents", JSONArray().put(
                    JSONObject().put("role", "user").put("parts", JSONArray().put(JSONObject().put("text", prompt)))
                )
            )
            val resp = http.newCall(request("gemini-flash-lite-latest", "generateContent", body)).await().okOrThrow()
            val out = resp.use { textOf(JSONObject(it.body!!.string())).trim() }
            if (out.isBlank() || out.equals("NONE", ignoreCase = true) || out.length > 140) null else out
        } catch (_: Exception) {
            null
        }
    }

class MeeraOverlayService : Service() {
    private var windowManager: WindowManager? = null
    private var edgeView: EdgeLightingView? = null

    companion object {
        @Volatile var isRunning = false
            private set

        fun start(ctx: Context) {
            if (isRunning) return
            if (!Settings.canDrawOverlays(ctx)) return
            try {
                ctx.startService(Intent(ctx, MeeraOverlayService::class.java))
            } catch (_: Exception) {}
        }

        fun stop(ctx: Context) {
            if (!isRunning) return
            try {
                ctx.stopService(Intent(ctx, MeeraOverlayService::class.java))
            } catch (_: Exception) {}
        }
    }

    override fun onCreate() {
        super.onCreate()
        if (!Settings.canDrawOverlays(this)) {
            stopSelf()
            return
        }
        isRunning = true
        windowManager = getSystemService(Context.WINDOW_SERVICE) as WindowManager

        val layoutType = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY
        } else {
            @Suppress("DEPRECATION")
            WindowManager.LayoutParams.TYPE_PHONE
        }

        val params = WindowManager.LayoutParams(
            WindowManager.LayoutParams.MATCH_PARENT,
            WindowManager.LayoutParams.MATCH_PARENT,
            layoutType,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                WindowManager.LayoutParams.FLAG_NOT_TOUCHABLE or
                WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.START
        }

        edgeView = EdgeLightingView(this)
        try {
            windowManager?.addView(edgeView, params)
        } catch (_: Exception) {}
    }

    override fun onDestroy() {
        isRunning = false
        try {
            edgeView?.let { windowManager?.removeView(it) }
            edgeView = null
        } catch (_: Exception) {}
        super.onDestroy()
    }

    override fun onBind(intent: Intent?): IBinder? = null
}

class EdgeLightingView(context: Context) : View(context) {
    private val strokePaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        style = Paint.Style.STROKE
        strokeWidth = 14f
    }
    private val glowPaint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        style = Paint.Style.STROKE
        strokeWidth = 28f
        alpha = 90
    }
    private val rect = RectF()
    private var hue = 0f

    override fun onDraw(canvas: Canvas) {
        super.onDraw(canvas)
        hue = (hue + 3.5f) % 360f

        val c1 = Color.HSVToColor(floatArrayOf(hue, 0.95f, 1f))
        val c2 = Color.HSVToColor(floatArrayOf((hue + 90f) % 360f, 0.95f, 1f))
        val c3 = Color.HSVToColor(floatArrayOf((hue + 180f) % 360f, 0.95f, 1f))
        val c4 = Color.HSVToColor(floatArrayOf((hue + 270f) % 360f, 0.95f, 1f))

        val shader = LinearGradient(
            0f, 0f, width.toFloat(), height.toFloat(),
            intArrayOf(c1, c2, c3, c4, c1),
            null, Shader.TileMode.CLAMP
        )
        strokePaint.shader = shader
        glowPaint.shader = shader

        val pad = 8f
        rect.set(pad, pad, width.toFloat() - pad, height.toFloat() - pad)
        canvas.drawRoundRect(rect, 38f, 38f, glowPaint)
        canvas.drawRoundRect(rect, 38f, 38f, strokePaint)

        postInvalidateOnAnimation()
    }
}

class MeeraNotificationService : NotificationListenerService() {
    companion object {
        val recentNotifs = CopyOnWriteArrayList<NotifItem>()
        fun getLatestSummary(): String {
            if (recentNotifs.isEmpty()) return "अभी कोई नया नोटिफिकेशन नहीं है।"
            return recentNotifs.takeLast(8).reversed().joinToString("\n") { "• [${it.title}]: ${it.text}" }
        }
    }

    override fun onNotificationPosted(sbn: StatusBarNotification?) {
        val n = sbn?.notification ?: return
        val pkg = sbn.packageName ?: return
        if (pkg == packageName) return
        val ex = n.extras ?: return
        val title = ex.getCharSequence("android.title")?.toString()?.trim().orEmpty()
        val text = ex.getCharSequence("android.text")?.toString()?.trim().orEmpty()
        if (title.isNotBlank() || text.isNotBlank()) {
            recentNotifs.add(NotifItem(pkg, title, text, System.currentTimeMillis()))
            while (recentNotifs.size > 25) recentNotifs.removeAt(0)
        }
    }

    override fun onNotificationRemoved(sbn: StatusBarNotification?) {}
}

class MeeraAccessibilityService : AccessibilityService() {

    companion object {
        @Volatile var instance: MeeraAccessibilityService? = null
            private set
        @Volatile var currentPackage: String = ""
            private set
    }

    override fun onServiceConnected() {
        super.onServiceConnected()
        instance = this
    }

    override fun onAccessibilityEvent(event: AccessibilityEvent?) {
        event?.packageName?.toString()?.let { if (it.isNotBlank()) currentPackage = it }
    }

    override fun onInterrupt() {}

    override fun onDestroy() {
        if (instance === this) instance = null
        super.onDestroy()
    }

    /** चालू स्क्रीन के सारे पठनीय टेक्स्ट, बटन और लेबल्स निकालना */
    fun readScreenText(maxItems: Int = 45): String {
        val root = rootInActiveWindow ?: return ""
        val items = LinkedHashSet<String>()
        val pkg = root.packageName?.toString().orEmpty()
        collectNodes(root, items, maxItems)
        if (items.isEmpty()) return ""
        return "Active App: $pkg\n" + items.joinToString("\n") { "• $it" }
    }

    private fun collectNodes(node: AccessibilityNodeInfo?, out: MutableSet<String>, limit: Int) {
        if (node == null || out.size >= limit) return
        if (node.isVisibleToUser) {
            val t = node.text?.toString()?.trim().orEmpty()
            val d = node.contentDescription?.toString()?.trim().orEmpty()
            val label = when {
                t.isNotBlank() && d.isNotBlank() && !t.equals(d, true) -> "$t ($d)"
                t.isNotBlank() -> t
                d.isNotBlank() -> d
                else -> ""
            }
            if (label.length in 2..180) {
                val tag = if (node.isClickable) "[Button] " else if (node.isEditable) "[Input] " else ""
                out.add("$tag$label")
            }
        }
        for (i in 0 until node.childCount) {
            if (out.size >= limit) break
            collectNodes(node.getChild(i), out, limit)
        }
    }

    /** स्क्रीन पर दिख रहे किसी टेक्स्ट या बटन को ढूँढकर उस पर क्लिक करना */
    fun clickByText(query: String): Boolean {
        val root = rootInActiveWindow ?: return false
        val target = findNodeByText(root, query.trim()) ?: return false
        return performClickOnNode(target)
    }

    private fun findNodeByText(node: AccessibilityNodeInfo?, q: String): AccessibilityNodeInfo? {
        if (node == null) return null
        if (node.isVisibleToUser) {
            val t = node.text?.toString()?.trim().orEmpty()
            val d = node.contentDescription?.toString()?.trim().orEmpty()
            if (t.contains(q, ignoreCase = true) || d.contains(q, ignoreCase = true)) {
                return node
            }
        }
        for (i in 0 until node.childCount) {
            val found = findNodeByText(node.getChild(i), q)
            if (found != null) return found
        }
        return null
    }

    private fun performClickOnNode(node: AccessibilityNodeInfo): Boolean {
        var cur: AccessibilityNodeInfo? = node
        while (cur != null) {
            if (cur.isClickable && cur.performAction(AccessibilityNodeInfo.ACTION_CLICK)) return true
            cur = cur.parent
        }
        val rect = Rect()
        node.getBoundsInScreen(rect)
        if (!rect.isEmpty) {
            val path = Path().apply { moveTo(rect.exactCenterX(), rect.exactCenterY()) }
            val stroke = GestureDescription.StrokeDescription(path, 0, 80)
            return dispatchGesture(GestureDescription.Builder().addStroke(stroke).build(), null, null)
        }
        return false
    }

    /** स्क्रीन के इनपुट बॉक्स (जैसे WhatsApp/Search बार) में टेक्स्ट टाइप करना */
    fun inputText(text: String, autoSend: Boolean = true): Boolean {
        val root = rootInActiveWindow ?: return false
        val editNode = findEditableNode(root) ?: return false
        editNode.performAction(AccessibilityNodeInfo.ACTION_FOCUS)
        val args = Bundle().apply {
            putCharSequence(AccessibilityNodeInfo.ACTION_ARGUMENT_SET_TEXT_CHARSEQUENCE, text)
        }
        val ok = editNode.performAction(AccessibilityNodeInfo.ACTION_SET_TEXT, args)
        if (ok && autoSend) {
            Thread.sleep(250)
            clickByText("Send") || clickByText("भेजें") || clickByText("Search")
        }
        return ok
    }

    private fun findEditableNode(node: AccessibilityNodeInfo?): AccessibilityNodeInfo? {
        if (node == null) return null
        if (node.isVisibleToUser && node.isEditable) return node
        for (i in 0 until node.childCount) {
            val f = findEditableNode(node.getChild(i))
            if (f != null) return f
        }
        return null
    }

    /** सिस्टम नेविगेशन और स्क्रॉल (home, back, notifications, scroll_down, scroll_up) */
    fun performSystemAction(action: String): Boolean {
        return when (action.lowercase().trim()) {
            "home" -> performGlobalAction(GLOBAL_ACTION_HOME)
            "back" -> performGlobalAction(GLOBAL_ACTION_BACK)
            "recents" -> performGlobalAction(GLOBAL_ACTION_RECENTS)
            "notifications" -> performGlobalAction(GLOBAL_ACTION_NOTIFICATIONS)
            "scroll_down", "down" -> scrollScreen(forward = true)
            "scroll_up", "up" -> scrollScreen(forward = false)
            else -> false
        }
    }

    private fun scrollScreen(forward: Boolean): Boolean {
        val root = rootInActiveWindow ?: return false
        val scrollable = findScrollable(root)
        if (scrollable != null) {
            val act = if (forward) AccessibilityNodeInfo.ACTION_SCROLL_FORWARD else AccessibilityNodeInfo.ACTION_SCROLL_BACKWARD
            if (scrollable.performAction(act)) return true
        }
        val dm = resources.displayMetrics
        val cx = dm.widthPixels / 2f
        val startY = if (forward) dm.heightPixels * 0.75f else dm.heightPixels * 0.25f
        val endY = if (forward) dm.heightPixels * 0.25f else dm.heightPixels * 0.75f
        val path = Path().apply { moveTo(cx, startY); lineTo(cx, endY) }
        val stroke = GestureDescription.StrokeDescription(path, 0, 260)
        return dispatchGesture(GestureDescription.Builder().addStroke(stroke).build(), null, null)
    }

    private fun findScrollable(node: AccessibilityNodeInfo?): AccessibilityNodeInfo? {
        if (node == null) return null
        if (node.isScrollable) return node
        for (i in 0 until node.childCount) {
            val f = findScrollable(node.getChild(i))
            if (f != null) return f
        }
        return null
    }
}

    /** Gemini से आए किसी भी Function Call को एग्जीक्यूट करना */
    fun executeFunctionCall(ctx: Context, name: String, args: JSONObject): ToolExecutionResult {
        return when (name) {
            "open_app" -> {
                val appName = args.optString("app_name", "").trim()
                val ok = openAppByName(ctx, appName)
                if (ok) ToolExecutionResult(true, name, "$appName खोल दिया है! बताओ इसमें आगे क्या करना है?", "SUCCESS: Opened app $appName")
                else ToolExecutionResult(true, name, "फोन में $appName नाम का ऐप नहीं मिला।", "ERROR: App $appName not found on device")
            }
            "read_screen" -> {
                val svc = MeeraAccessibilityService.instance
                if (svc == null) {
                    ToolExecutionResult(
                        true, name,
                        "स्क्रीन पढ़ने के लिए पहले Accessibility सर्विस ऑन करनी होगी।",
                        "ERROR: Accessibility service is OFF. Ask user to enable Meera AI in Accessibility Settings."
                    )
                } else {
                    val content = svc.readScreenText()
                    val summary = if (content.isBlank()) "स्क्रीन पर अभी कोई टेक्स्ट नहीं दिख रहा है।" else content
                    ToolExecutionResult(true, name, "", "CURRENT SCREEN CONTENT:\n$summary")
                }
            }
            "screen_action" -> {
                val act = args.optString("action_type", "").lowercase().trim()
                val target = args.optString("target_or_text", "").trim()
                val svc = MeeraAccessibilityService.instance
                if (svc == null) {
                    ToolExecutionResult(true, name, "यह काम करने के लिए कृपया Accessibility परमिशन ऑन करें।", "ERROR: Accessibility Service OFF")
                } else {
                    val ok = when (act) {
                        "click" -> svc.clickByText(target)
                        "type" -> svc.inputText(target, autoSend = true)
                        else -> svc.performSystemAction(act)
                    }
                    ToolExecutionResult(true, name, if (ok) "कर दिया।" else "स्क्रीन पर '$target' नहीं मिला।", if (ok) "SUCCESS: $act ($target)" else "FAILED: $act ($target)")
                }
            }
            "whatsapp_message" -> {
                val person = args.optString("contact_name", "").trim()
                val msg = args.optString("message", "").trim()
                sendWhatsAppToPerson(ctx, person, msg)
            }
            "make_phone_call" -> {
                val person = args.optString("contact_name", "").trim()
                callPerson(ctx, person)
            }
            "play_or_search" -> {
                val q = args.optString("query", "").trim()
                val platform = args.optString("platform", "youtube").lowercase()
                playOrSearchMedia(ctx, q, platform)
            }
            else -> ToolExecutionResult(false)
        }
    }

    /**
     * लोकल फास्ट कमांड राउटर (Modular/Chat मोड में बिना रुके तुरंत ऐप खोलने या स्क्रीन पढ़ने के लिए)
     */
    fun tryDirectVoiceCommand(ctx: Context, userText: String): ToolExecutionResult {
        val raw = userText.trim()
        val low = raw.lowercase()

        // 1. स्क्रीन पढ़ने या नोटिफिकेशन पढ़ने की कमांड
        if (low.contains("स्क्रॉल डाउन") || low.contains("नीचे स्क्रॉल") || low.contains("नीचे करो") || low.contains("scroll down")) {
            val ok = MeeraAccessibilityService.instance?.performSystemAction("scroll_down") == true
            return ToolExecutionResult(true, "screen_action", if (ok) "नीचे स्क्रॉल कर दिया है। और नीचे करूँ या कुछ खोलना है?" else "स्क्रॉल करने के लिए Accessibility ऑन करें।", "SUCCESS: Scrolled Down")
        }
        if (low.contains("स्क्रॉल अप") || low.contains("ऊपर स्क्रॉल") || low.contains("ऊपर करो") || low.contains("scroll up")) {
            val ok = MeeraAccessibilityService.instance?.performSystemAction("scroll_up") == true
            return ToolExecutionResult(true, "screen_action", if (ok) "ऊपर स्क्रॉल कर दिया है। आगे बताओ क्या करना है?" else "Accessibility ऑन करें।", "SUCCESS: Scrolled Up")
        }
        if (low == "बैक करो" || low == "पीछे जाओ" || low == "वापस आओ" || low == "go back" || low == "back") {
            val ok = MeeraAccessibilityService.instance?.performSystemAction("back") == true
            return ToolExecutionResult(true, "screen_action", if (ok) "पीछे आ गई। अब बताओ क्या करें?" else "Accessibility ऑन करें।", "SUCCESS: Back")
        }
        if (low.contains("स्क्रीन पढ़") || low.contains("screen padh") || low.contains("स्क्रीन पर क्या") ||
            low.contains("screen par kya") || low.contains("read screen")
        ) {
            return executeFunctionCall(ctx, "read_screen", JSONObject())
        }

        if (low.contains("नोटिफिकेशन") || low.contains("notification") || low.contains("किसका मैसेज आया")) {
            val notifSummary = MeeraNotificationService.getLatestSummary()
            return ToolExecutionResult(true, "read_notifications", "", "RECENT NOTIFICATIONS:\n$notifSummary")
        }

        // 2. होम स्क्रीन या बैक जाने की कमांड
        if (low == "होम स्क्रीन पे आओ" || low == "होम स्क्रीन पर आओ" || low == "go home" || low == "home screen") {
            val svc = MeeraAccessibilityService.instance
            if (svc != null && svc.performSystemAction("home")) {
                return ToolExecutionResult(true, "screen_action", "आ गई होम स्क्रीन पर, बताइए अब क्या करना है?", "SUCCESS: Navigated Home")
            }
        }

        // 3. YouTube पर गाना/वीडियो चलाने की कमांड
        val ytMatch = Regex("(?:youtube\\s*(?:पर|पे|par|pe|me)|यूट्यूब\\s*(?:पर|पे))\\s+(.+?)(?:\\s+(?:चलाओ|चला दो|प्ले करो|खोलो|लगाओ|play|search))?$", RegexOption.IGNORE_CASE).find(raw)
        if (ytMatch != null) {
            val song = ytMatch.groupValues[1].trim()
            if (song.isNotBlank()) return playOrSearchMedia(ctx, song, "youtube")
        }

        // 4. WhatsApp मैसेज कमांड (जैसे: "लक्ष्मण को व्हाट्सएप पर मैसेज भेजो कि...")
        val waMatch = Regex("(.+?)\\s+को\\s+(?:whatsapp|व्हाट्सएप|व्हाट्सऐप).*?(?:मैसेज|message)\\s+(?:भेजो|करो|लिखो)(?:\\s+कि|\\s+ki)?\\s*(.*)", RegexOption.IGNORE_CASE).find(raw)
        if (waMatch != null) {
            val person = waMatch.groupValues[1].replace(Regex("^(?:मीरा|अरे|सुनो|एक काम करो)\\s*"), "").trim()
            val msg = waMatch.groupValues[2].trim()
            return sendWhatsAppToPerson(ctx, person, msg)
        }

        // 5. कॉल लगाने की कमांड (जैसे: "लक्ष्मण को कॉल लगाओ / फोन करो")
        val callMatch = Regex("(.+?)\\s+को\\s+(?:कॉल|फोन|call|phone)\\s+(?:लगाओ|करो|मिलाओ|लगा दो)", RegexOption.IGNORE_CASE).find(raw)
        if (callMatch != null) {
            val person = callMatch.groupValues[1].replace(Regex("^(?:मीरा|अरे|सुनो)\\s*"), "").trim()
            return callPerson(ctx, person)
        }

        // 6. कोई भी ऐप खोलने की कमांड (जैसे: "WhatsApp ओपन करो", "यूट्यूब खोलो", "open instagram")
        val openPatterns = listOf(
            Regex("^(?:मीरा|अरे|सुनो|प्लीज)?\\s*(.+?)\\s+(?:ओपन करो|ओपन कर दो|खोलो|खोल दो|चालू करो|open karo|open kar do|kholo|khol do)$", RegexOption.IGNORE_CASE),
            Regex("^(?:open|launch)\\s+(.+)$", RegexOption.IGNORE_CASE)
        )
        for (pat in openPatterns) {
            val m = pat.find(raw)
            if (m != null) {
                val appCandidate = m.groupValues[1]
                    .replace(Regex("(?:ऐप|app|एप्लीकेशन|application)$", RegexOption.IGNORE_CASE), "")
                    .trim()
                if (appCandidate.length in 2..30 && openAppByName(ctx, appCandidate)) {
                    return ToolExecutionResult(true, "open_app", "$appCandidate खोल दिया है।", "SUCCESS: Opened $appCandidate")
                }
            }
        }

        return ToolExecutionResult(false)
    }

    /** फोन में इंस्टॉल किसी भी ऐप को उसके नाम (हिंदी या इंग्लिश) से खोलना */
    fun openAppByName(ctx: Context, rawName: String): Boolean {
        val clean = rawName.lowercase().replace("ऐप", "").replace("app", "").trim()
        if (clean.isBlank()) return false

        // स्पेशल सिस्टम स्क्रीन
        if (clean == "settings" || clean == "सेटिंग" || clean == "सेटिंग्स") {
            ctx.startActivity(Intent(Settings.ACTION_SETTINGS).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK))
            return true
        }

        val pm = ctx.packageManager

        // 1. पहले Known Apps मैप में चेक करें
        for ((key, pkgs) in KNOWN_APPS) {
            if (clean.contains(key) || key.contains(clean)) {
                for (pkg in pkgs) {
                    val launch = pm.getLaunchIntentForPackage(pkg)
                    if (launch != null) {
                        launch.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                        ctx.startActivity(launch)
                        return true
                    }
                }
            }
        }

        // 2. फोन के सारे Launcher Apps के नामों में मैच ढूँढें
        return try {
            val mainIntent = Intent(Intent.ACTION_MAIN, null).addCategory(Intent.CATEGORY_LAUNCHER)
            val resolved = pm.queryIntentActivities(mainIntent, 0)
            for (info in resolved) {
                val label = info.loadLabel(pm).toString().lowercase()
                val pkg = info.activityInfo.packageName.lowercase()
                if (label == clean || label.contains(clean) || clean.contains(label) || pkg.contains(clean)) {
                    val launch = pm.getLaunchIntentForPackage(info.activityInfo.packageName)
                    if (launch != null) {
                        launch.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                        ctx.startActivity(launch)
                        return true
                    }
                }
            }
            false
        } catch (_: Exception) {
            false
        }
    }

    /** कॉन्टैक्ट्स में से नाम के आधार पर मोबाइल नंबर ढूँढना */
    fun findContactNumber(ctx: Context, nameQuery: String): Pair<String, String>? {
        val hasPerm = ContextCompat.checkSelfPermission(ctx, Manifest.permission.READ_CONTACTS) == PackageManager.PERMISSION_GRANTED
        if (!hasPerm || nameQuery.isBlank()) return null
        val q = nameQuery.trim().lowercase()
        var cursor: Cursor? = null
        try {
            cursor = ctx.contentResolver.query(
                ContactsContract.CommonDataKinds.Phone.CONTENT_URI,
                arrayOf(ContactsContract.CommonDataKinds.Phone.DISPLAY_NAME, ContactsContract.CommonDataKinds.Phone.NUMBER),
                null, null, null
            )
            if (cursor != null) {
                var bestMatch: Pair<String, String>? = null
                while (cursor.moveToNext()) {
                    val dName = cursor.getString(0) ?: continue
                    val num = cursor.getString(1) ?: continue
                    if (dName.equals(q, ignoreCase = true)) return dName to num
                    if (dName.lowercase().contains(q) || q.contains(dName.lowercase())) {
                        bestMatch = dName to num
                    }
                }
                return bestMatch
            }
        } catch (_: Exception) {
        } finally {
            cursor?.close()
        }
        return null
    }

    private fun sendWhatsAppToPerson(ctx: Context, person: String, message: String): ToolExecutionResult {
        val contact = findContactNumber(ctx, person)
        val rawDigits = (contact?.second ?: person).replace(Regex("[^0-9+]"), "")
        val phone = when {
            rawDigits.length == 10 -> "91$rawDigits"
            rawDigits.startsWith("+") -> rawDigits.substring(1)
            else -> rawDigits
        }

        return try {
            if (phone.length >= 10) {
                val encoded = URLEncoder.encode(message, "UTF-8")
                val uri = Uri.parse("https://api.whatsapp.com/send?phone=$phone" + if (message.isNotBlank()) "&text=$encoded" else "")
                val i = Intent(Intent.ACTION_VIEW, uri).apply {
                    setPackage("com.whatsapp")
                    addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
                }
                ctx.startActivity(i)
                // अगर Accessibility चालू है और मैसेज दिया गया है, तो 1.2s बाद Send बटन भी दबाने की कोशिश करें
                if (message.isNotBlank()) {
                    Thread {
                        try {
                            Thread.sleep(1400)
                            MeeraAccessibilityService.instance?.let { svc ->
                                svc.clickByText("Send") || svc.clickByText("भेजें")
                            }
                        } catch (_: Exception) {}
                    }.start()
                }
                val targetName = contact?.first ?: person
                ToolExecutionResult(true, "whatsapp_message", "$targetName की WhatsApp चैट खोल दी है।", "SUCCESS: Opened WhatsApp chat for $targetName")
            } else {
                // अगर कॉन्टैक्ट नंबर नहीं मिला, तो सीधे WhatsApp खोलें
                openAppByName(ctx, "WhatsApp")
                ToolExecutionResult(
                    true, "whatsapp_message",
                    "कॉन्टैक्ट्स में '$person' का नंबर नहीं मिला, इसलिए मैंने WhatsApp खोल दिया है।",
                    "Opened WhatsApp (contact '$person' not found in phonebook)"
                )
            }
        } catch (_: Exception) {
            openAppByName(ctx, "WhatsApp")
            ToolExecutionResult(true, "whatsapp_message", "WhatsApp खोल दिया है।", "Opened WhatsApp")
        }
    }

    private fun callPerson(ctx: Context, person: String): ToolExecutionResult {
        val contact = findContactNumber(ctx, person)
        val number = contact?.second ?: person.filter { it.isDigit() || it == '+' }
        if (number.length < 3) {
            return ToolExecutionResult(true, "make_phone_call", "कॉन्टैक्ट्स में '$person' का नंबर नहीं मिला।", "ERROR: Contact $person not found")
        }
        val canDirectCall = ContextCompat.checkSelfPermission(ctx, Manifest.permission.CALL_PHONE) == PackageManager.PERMISSION_GRANTED
        val action = if (canDirectCall) Intent.ACTION_CALL else Intent.ACTION_DIAL
        return try {
            val intent = Intent(action, Uri.parse("tel:$number")).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            ctx.startActivity(intent)
            val label = contact?.first ?: person
            ToolExecutionResult(true, "make_phone_call", "$label को कॉल लगा रही हूँ।", "SUCCESS: Calling $label ($number)")
        } catch (e: Exception) {
            ToolExecutionResult(true, "make_phone_call", "कॉल लगाने में दिक्कत आई।", "ERROR: ${e.message}")
        }
    }

    private fun playOrSearchMedia(ctx: Context, query: String, platform: String): ToolExecutionResult {
        return try {
            val encoded = URLEncoder.encode(query, "UTF-8")
            val uri = if (platform.contains("google")) {
                Uri.parse("https://www.google.com/search?q=$encoded")
            } else {
                Uri.parse("https://www.youtube.com/results?search_query=$encoded")
            }
            val i = Intent(Intent.ACTION_VIEW, uri).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            ctx.startActivity(i)
            ToolExecutionResult(true, "play_or_search", "YouTube पर '$query' लगा दिया है! सुनिए, और कुछ काम हो तो बताइए?", "SUCCESS: Opened $platform for $query")
        } catch (_: Exception) {
            ToolExecutionResult(false)
        }
    }
}

        roleHeld || cur.contains(ctx.packageName, ignoreCase = true)
    } catch (_: Exception) { false }

    fun openBatteryOptimization(ctx: Context) {
        try {
            val i = Intent(Settings.ACTION_REQUEST_IGNORE_BATTERY_OPTIMIZATIONS).apply {
                data = Uri.parse("package:${ctx.packageName}")
                addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            }
            ctx.startActivity(i)
        } catch (_: Exception) {
            try {
                ctx.startActivity(Intent(Settings.ACTION_IGNORE_BATTERY_OPTIMIZATION_SETTINGS).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK))
            } catch (_: Exception) {}
        }
    }

    fun openOverlaySettings(ctx: Context) {
        try {
            val i = Intent(Settings.ACTION_MANAGE_OVERLAY_PERMISSION, Uri.parse("package:${ctx.packageName}"))
                .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
            ctx.startActivity(i)
        } catch (_: Exception) {}
    }

    fun openNotificationListenerSettings(ctx: Context) {
        try {
            ctx.startActivity(Intent(Settings.ACTION_NOTIFICATION_LISTENER_SETTINGS).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK))
        } catch (_: Exception) {}
    }

    fun openAccessibilitySettings(ctx: Context) {
        try {
            ctx.startActivity(Intent(Settings.ACTION_ACCESSIBILITY_SETTINGS).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK))
        } catch (_: Exception) {}
    }

    fun openDefaultAssistantSettings(ctx: Context) {
        try {
            ctx.startActivity(Intent(Settings.ACTION_VOICE_INPUT_SETTINGS).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK))
        } catch (_: Exception) {
            try {
                ctx.startActivity(Intent(Settings.ACTION_MANAGE_DEFAULT_APPS_SETTINGS).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK))
            } catch (_: Exception) {}
        }
    }
}

@Composable
fun Orb(state: VState, vm: VoiceViewModel, modifier: Modifier = Modifier) {
    val lvlRaw by vm.level.collectAsStateWithLifecycle()
    val lvl by animateFloatAsState(lvlRaw.coerceIn(0f, 1f), tween(90), label = "lvl")
    val target = when (state) {
        VState.IDLE -> AccentBlue
        VState.CONNECTING -> Color(0xFF7C8CB8)
        VState.LISTENING -> Color(0xFF3CC8FF)
        VState.USER_SPEAKING -> AccentCyan
        VState.PROCESSING -> Color(0xFFFFB84D)
        VState.ASSISTANT_SPEAKING -> Color(0xFFB46BFF)
        VState.ERROR -> Color(0xFFE5484D)
    }
    val color by animateColorAsState(target, tween(500), label = "color")
    val t = rememberInfiniteTransition(label = "orb")
    val breathe by t.animateFloat(0f, 1f, infiniteRepeatable(tween(2600, easing = LinearEasing), RepeatMode.Reverse), label = "b")
    val spin by t.animateFloat(0f, 360f, infiniteRepeatable(tween(7000, easing = LinearEasing)), label = "s")
    val ring by t.animateFloat(0f, 1f, infiniteRepeatable(tween(1600, easing = LinearOutSlowInEasing)), label = "r")
    val rippling = state == VState.USER_SPEAKING || state == VState.ASSISTANT_SPEAKING

    Canvas(modifier) {
        val c = center
        val base = size.minDimension / 2f * 0.55f
        val r = base * (1f + 0.05f * breathe + 0.45f * lvl)
        drawCircle(Brush.radialGradient(listOf(color.copy(alpha = 0.40f), Color.Transparent), center = c, radius = r * 2f), radius = r * 2f, center = c)
        if (rippling) {
            for (i in 0..1) {
                val p = (ring + i * 0.5f) % 1f
                drawCircle(color.copy(alpha = (1f - p) * 0.4f), radius = r * (1f + p * 0.6f), center = c, style = Stroke(3.dp.toPx()))
            }
        }
        drawCircle(
            Brush.radialGradient(
                listOf(Color.White.copy(alpha = 0.85f), color, color.copy(alpha = 0.55f), Bg),
                center = c + Offset(-r * 0.3f, -r * 0.3f), radius = r * 1.5f
            ),
            radius = r, center = c
        )
        val arcs = if (state == VState.PROCESSING) 3 else 1
        for (i in 0 until arcs) {
            rotate(spin * (if (state == VState.PROCESSING) 2f else 1f) + i * 120f, pivot = c) {
                drawArc(
                    Color.White.copy(alpha = 0.28f), startAngle = 0f, sweepAngle = 60f, useCenter = false,
                    topLeft = c - Offset(r * 0.8f, r * 0.8f), size = Size(r * 1.6f, r * 1.6f),
                    style = Stroke(r * 0.1f, cap = StrokeCap.Round)
                )
            }
        }
    }
}

    fun sendTextQuery(query: String, speakAloud: Boolean = false) {
        val clean = query.trim()
        if (clean.isBlank()) return
        if (prefs.apiKey.isBlank()) {
            _ui.update { it.copy(askKey = true, error = "पहले API Key डालें") }
            return
        }
        viewModelScope.launch(Dispatchers.IO) {
            val startAt = System.currentTimeMillis()
            _ui.update { it.copy(user = clean, ai = "सोच रही हूँ…", error = null) }
            appendChatLog("user", clean)
            autoLearnFromUser(clean)

            // 1. पहले चेक करें कि क्या यह सीधा फोन टूल कमांड है (जैसे WhatsApp खोलो, स्क्रीन पढ़ो)
            val toolRes = MeeraTools.tryDirectVoiceCommand(appCtx, clean)
            if (toolRes.executed && toolRes.spokenReply.isNotBlank() && toolRes.rawDataForAi.startsWith("SUCCESS:")) {
                _ui.update {
                    it.copy(
                        ai = toolRes.spokenReply,
                        lastToolStatus = "⚡ ${toolRes.toolName}",
                        latencyMs = System.currentTimeMillis() - startAt
                    )
                }
                appendChatLog("model", toolRes.spokenReply)
                return@launch
            }

            val effectivePrompt = if (toolRes.executed && toolRes.rawDataForAi.isNotBlank()) {
                "$clean\n\n[TOOL OUTPUT]:\n${toolRes.rawDataForAi}"
            } else clean

            history.add("user" to effectivePrompt)
            val sysPrompt = SystemPrompt.build(prefs.persona, prefs.girlfriendMode, prefs.getMemories())
            val full = StringBuilder()
            try {
                rest.streamChat(history.takeLast(12), sysPrompt, prefs.chatModel) { s ->
                    full.append(s).append(' ')
                    _ui.update {
                        it.copy(
                            ai = full.toString().trim(),
                            latencyMs = System.currentTimeMillis() - startAt
                        )
                    }
                }
                val reply = full.toString().trim()
                if (reply.isNotBlank()) {
                    history.add("model" to reply)
                    appendChatLog("model", reply)
                    if (speakAloud) {
                        val pcm = rest.tts(reply.replace(Regex("[*#_`~]"), ""), prefs.voice, prefs.ttsModel, prefs.persona, prefs.girlfriendMode)
                        val tempPlayer = player ?: PcmPlayer().also { player = it }
                        tempPlayer.enqueue(pcm)
                    }
                }
            } catch (e: Exception) {
                val err = friendly(e)
                _ui.update { it.copy(ai = err, error = err) }
            }
        }
    }

    private fun onFrame(frame: ShortArray) {
        if (!active) return
        val rms = rmsOf(frame)
        val p = player
        val assistant = st == VState.ASSISTANT_SPEAKING || p?.busy == true
        _level.value = if (assistant) (p?.level ?: 0f) else (rms * 6f).coerceIn(0f, 1f)

        if (assistant) {
            if (rms > Cfg.BARGE_RMS) {
                bargeFrames++
                if (bargeFrames >= Cfg.BARGE_FRAMES) { bargeFrames = 0; bargeIn() }
            } else bargeFrames = 0
        } else bargeFrames = 0

        if (st == VState.ASSISTANT_SPEAKING) {
            if (p?.busy != true && !turnOpen()) {
                quietMs += Cfg.FRAME_MS
                if (quietMs >= Cfg.ASSISTANT_QUIET_MS) {
                    quietMs = 0
                    detector.reset()
                    setState(VState.LISTENING)
                }
            } else quietMs = 0
        }

        val busy = st == VState.ASSISTANT_SPEAKING || player?.busy == true
        if (prefs.mode == Mode.LIVE) liveFrame(frame, rms, busy) else modularFrame(frame, rms, busy)
    }

    private fun bargeIn() {
        player?.flush()
        if (prefs.mode == Mode.MODULAR) {
            pipeline?.cancel()
            requestInFlight.set(false)
            utter.clear(); utter.addAll(preRoll); collecting = true
        } else {
            dropAudio = true
            liveTurnOpen = false
        }
        detector.forceSpeaking()
        quietMs = 0
        setState(VState.USER_SPEAKING)
    }

    private fun liveFrame(frame: ShortArray, rms: Float, busy: Boolean) {
        val l = live ?: return
        if (!l.ready) return
        val send = if (busy && rms < Cfg.BARGE_RMS) ShortArray(frame.size) else frame
        l.sendAudio(send.toLeBytes())
        if (busy) return
        when (detector.process(rms)) {
            TurnDetector.Ev.SPEECH_START -> if (st == VState.LISTENING || st == VState.PROCESSING) setState(VState.USER_SPEAKING)
            TurnDetector.Ev.SPEECH_END -> {
                speechEndAt = System.currentTimeMillis(); firstAudioPending = true
                if (st == VState.USER_SPEAKING) setState(VState.PROCESSING)
            }
            TurnDetector.Ev.NOISE -> if (st == VState.USER_SPEAKING) setState(VState.LISTENING)
            TurnDetector.Ev.NONE -> {}
        }
        if (st == VState.PROCESSING && System.currentTimeMillis() - speechEndAt > 8000) setState(VState.LISTENING)
    }

    private val liveEvents = object : GeminiLive.Events {
        override fun onReady() {
            try { startMic() } catch (e: Exception) { fail(e.message ?: "माइक शुरू नहीं हुआ") }
        }

        override fun onAudio(pcm: ByteArray) {
            if (dropAudio || !active) return
            liveTurnOpen = true
            if (st != VState.ASSISTANT_SPEAKING) {
                if (firstAudioPending) {
                    firstAudioPending = false
                    val ms = System.currentTimeMillis() - speechEndAt
                    _ui.update { it.copy(latencyMs = ms) }
                }
                setState(VState.ASSISTANT_SPEAKING)
            }
            player?.enqueue(pcm)
        }

        override fun onInputText(t: String) {
            dropAudio = false
            if (newUserTurn) { userBuf.clear(); aiBuf.clear(); newUserTurn = false }
            userBuf.append(t)
            _ui.update { it.copy(user = userBuf.toString().trim(), ai = "") }
        }

        override fun onOutputText(t: String) {
            if (dropAudio) return
            aiBuf.append(t)
            _ui.update { it.copy(ai = aiBuf.toString().trim()) }
        }

        override fun onToolCall(id: String, name: String, args: JSONObject) {
            viewModelScope.launch(Dispatchers.IO) {
                val res = MeeraTools.executeFunctionCall(appCtx, name, args)
                _ui.update { it.copy(lastToolStatus = "⚡ $name") }
                live?.sendToolResponse(id, name, res.rawDataForAi.ifBlank { res.spokenReply })
            }
        }

        override fun onInterrupted() {
            player?.flush()
            dropAudio = false; liveTurnOpen = false; newUserTurn = true
            if (st == VState.ASSISTANT_SPEAKING || st == VState.PROCESSING) {
                detector.forceSpeaking()
                setState(VState.USER_SPEAKING)
            }
        }

        override fun onTurnComplete() {
            val finishedUserText = userBuf.toString().trim()
            val finishedAiText = aiBuf.toString().trim()
            if (finishedUserText.isNotBlank()) {
                appendChatLog("user", finishedUserText)
                autoLearnFromUser(finishedUserText)
                // अगर मॉडल ने टूल कॉल किए बिना बोला हो लेकिन यूज़र ने ऐप खोलने कहा हो, तो बैकअप के तौर पर चलाएँ
                MeeraTools.tryDirectVoiceCommand(appCtx, finishedUserText)
            }
            if (finishedAiText.isNotBlank()) appendChatLog("model", finishedAiText)
            liveTurnOpen = false; newUserTurn = true; dropAudio = false
        }

        override fun onError(msg: String) {
            if (active) fail(msg)
        }
    }

    private fun modularFrame(frame: ShortArray, rms: Float, busy: Boolean) {
        if (!collecting) {
            preRoll.addLast(frame)
            while (preRoll.size > Cfg.PREROLL_FRAMES) preRoll.removeFirst()
        }
        if (busy || st == VState.PROCESSING || st == VState.CONNECTING || requestInFlight.get()) return
        if (collecting) utter.add(frame)
        when (detector.process(rms)) {
            TurnDetector.Ev.SPEECH_START -> {
                utter.clear(); utter.addAll(preRoll); collecting = true
                setState(VState.USER_SPEAKING)
            }
            TurnDetector.Ev.SPEECH_END -> {
                collecting = false
                val endAt = System.currentTimeMillis()
                val wav = pcmToWav(joinPcm(utter), Cfg.IN_RATE)
                utter.clear(); preRoll.clear()
                runPipeline(wav, endAt)
            }
            TurnDetector.Ev.NOISE -> {
                collecting = false; utter.clear()
                if (st == VState.USER_SPEAKING) setState(VState.LISTENING)
            }
            TurnDetector.Ev.NONE -> {}
        }
    }

    private fun runPipeline(wav: ByteArray, endAt: Long) {
        if (!requestInFlight.compareAndSet(false, true)) return
        pipeline?.cancel()
        setState(VState.PROCESSING)
        pipeline = viewModelScope.launch(Dispatchers.IO) {
            val full = StringBuilder()
            var userAdded = false
            try {
                val text = rest.transcribe(wav, prefs.sttModel)
                if (text.isBlank()) {
                    requestInFlight.set(false)
                    setState(VState.LISTENING)
                    return@launch
                }
                _ui.update { it.copy(user = text, ai = "", latencyMs = null) }
                appendChatLog("user", text)
                autoLearnFromUser(text)

                // Modular डायरेक्ट कमांड चेक (जैसे "WhatsApp खोलो", "स्क्रीन पढ़ो")
                val toolRes = MeeraTools.tryDirectVoiceCommand(appCtx, text)
                if (toolRes.executed && toolRes.spokenReply.isNotBlank() && toolRes.rawDataForAi.startsWith("SUCCESS:")) {
                    _ui.update { it.copy(ai = toolRes.spokenReply, lastToolStatus = "⚡ ${toolRes.toolName}") }
                    appendChatLog("model", toolRes.spokenReply)
                    val pcm = rest.tts(toolRes.spokenReply, prefs.voice, prefs.ttsModel, prefs.persona, prefs.girlfriendMode)
                    _ui.update { it.copy(latencyMs = System.currentTimeMillis() - endAt) }
                    setState(VState.ASSISTANT_SPEAKING)
                    player?.enqueue(pcm)
                    return@launch
                }

                val effectiveUserMsg = if (toolRes.executed && toolRes.rawDataForAi.isNotBlank()) {
                    "$text\n\n[SCREEN / TOOL DATA]:\n${toolRes.rawDataForAi}"
                } else text

                history.add("user" to effectiveUserMsg); userAdded = true

                val sysPrompt = SystemPrompt.build(prefs.persona, prefs.girlfriendMode, prefs.getMemories())
                val ttsJobs = Channel<Deferred<ByteArray>>(capacity = 3)
                val voice = prefs.voice
                val selectedTts = prefs.ttsModel
                val curPersona = prefs.persona
                val curGf = prefs.girlfriendMode
                var first = true

                val speaker = launch {
                    for (job in ttsJobs) {
                        val pcm = job.await()
                        if (first) {
                            first = false
                            val ms = System.currentTimeMillis() - endAt
                            _ui.update { it.copy(latencyMs = ms) }
                            setState(VState.ASSISTANT_SPEAKING)
                        }
                        player?.enqueue(pcm)
                    }
                }
                try {
                    rest.streamChat(history.takeLast(12), sysPrompt, prefs.chatModel) { s ->
                        full.append(s).append(' ')
                        _ui.update { it.copy(ai = full.toString().trim()) }
                        val cleanText = s.replace(Regex("[*#_`~]"), "").trim()
                        if (cleanText.isNotEmpty()) {
                            val job = async(Dispatchers.IO) {
                                rest.tts(cleanText, voice, selectedTts, curPersona, curGf)
                            }
                            ttsJobs.send(job)
                        }
                    }
                } finally {
                    ttsJobs.close()
                }
                speaker.join()
            } catch (e: CancellationException) {
                throw e
            } catch (e: Exception) {
                if (active) fail(friendly(e))
            } finally {
                requestInFlight.set(false)
                if (userAdded) {
                    val reply = full.toString().trim()
                    if (reply.isNotBlank()) {
                        history.add("model" to reply)
                        appendChatLog("model", reply)
                    } else history.removeLastOrNull()
                }
            }
        }
    }

    private fun friendly(e: Throwable): String {
        val m = e.message ?: e.javaClass.simpleName
        return when {
            e is UnknownHostException -> "इंटरनेट कमज़ोर या बंद है"
            m.contains("API key", true) || m.contains("HTTP 401") || m.contains("HTTP 403") -> "API Key सही नहीं है या इजाज़त नहीं: ${m.take(120)}"
            m.contains("HTTP 429") -> "कोटा/लिमिट पूरी हो गई (429)"
            else -> m.take(160)
        }
    }

    override fun onCleared() {
        stop()
        super.onCleared()
    }
}
